/**
 * Paystack Payment Integration
 * Handles payment processing and verification
 */

const PAYSTACK_BASE_URL = 'https://api.paystack.co';
const PAYSTACK_SECRET_KEY = process.env.PAYSTACK_SECRET_KEY;

/**
 * Initialize payment transaction
 * @param email - Customer email
 * @param amount - Amount in kobo (NGN * 100)
 * @param metadata - Additional metadata
 * @returns Authorization URL and reference
 */
export async function initializePayment(
  email: string,
  amount: number,
  metadata?: Record<string, any>
) {
  try {
    const response = await fetch(`${PAYSTACK_BASE_URL}/transaction/initialize`, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${PAYSTACK_SECRET_KEY}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        email,
        amount,
        metadata,
      }),
    });

    if (!response.ok) {
      throw new Error('Failed to initialize payment');
    }

    const data = await response.json();
    return data;
  } catch (error) {
    console.error('Error initializing payment:', error);
    throw error;
  }
}

/**
 * Verify payment transaction
 * @param reference - Payment reference from Paystack
 * @returns Transaction details
 */
export async function verifyPayment(reference: string) {
  try {
    const response = await fetch(
      `${PAYSTACK_BASE_URL}/transaction/verify/${reference}`,
      {
        method: 'GET',
        headers: {
          Authorization: `Bearer ${PAYSTACK_SECRET_KEY}`,
        },
      }
    );

    if (!response.ok) {
      throw new Error('Failed to verify payment');
    }

    const data = await response.json();
    return data;
  } catch (error) {
    console.error('Error verifying payment:', error);
    throw error;
  }
}

/**
 * Validate Paystack webhook signature
 * @param body - Request body
 * @param signature - X-Paystack-Signature header
 * @returns Whether signature is valid
 */
export function validateWebhookSignature(
  body: string,
  signature: string
): boolean {
  try {
    const crypto = require('crypto');
    const hash = crypto
      .createHmac('sha512', PAYSTACK_SECRET_KEY)
      .update(body)
      .digest('hex');

    return hash === signature;
  } catch (error) {
    console.error('Error validating webhook signature:', error);
    return false;
  }
}

import crypto from 'crypto';
import type { PaymentAdapter, InitializePaymentParams, InitializePaymentResult, VerifyPaymentResult, ValidateWebhookResult } from './adapter';

const BASE = 'https://api.paystack.co';
const SECRET = process.env.PAYSTACK_SECRET_KEY!;

export const PaystackAdapter: PaymentAdapter = {
  name: 'paystack',

  async initialize({ email, amount, metadata, currency = 'USD' }): Promise<InitializePaymentResult> {
    const res = await fetch(BASE + '/transaction/initialize', {
      method: 'POST',
      headers: { Authorization: 'Bearer ' + SECRET, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        email,
        amount,
        metadata,
        currency,
        channels: ['card', 'apple_pay'],
      }),
    });
    const data = await res.json();
    if (!data.status) throw new Error(data.message || 'Paystack init failed');
    return {
      authorizationUrl: data.data.authorization_url,
      reference: data.data.reference,
      provider: 'paystack',
    };
  },

  async verify(reference): Promise<VerifyPaymentResult> {
    const res = await fetch(BASE + '/transaction/verify/' + reference, {
      headers: { Authorization: 'Bearer ' + SECRET },
    });
    const data = await res.json();
    const tx = data.data;
    return {
      success: tx?.status === 'success',
      reference: tx?.reference || reference,
      amount: tx?.amount || 0,
      metadata: tx?.metadata || {},
      provider: 'paystack',
    };
  },

  validateWebhook(body, signature): ValidateWebhookResult {
    try {
      const hash = crypto.createHmac('sha512', SECRET).update(body).digest('hex');
      if (hash !== signature) return { valid: false, event: '', data: null };
      const parsed = JSON.parse(body);
      return { valid: true, event: parsed.event, data: parsed.data };
    } catch {
      return { valid: false, event: '', data: null };
    }
  },
};
